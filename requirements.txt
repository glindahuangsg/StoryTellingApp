# requirements.txt — Python dependencies for "Story Time!"

# IMPORTANT: the --extra-index-url line below makes pip fetch the CPU-only build of
# PyTorch (~200MB) instead of the default PyPI wheel (which bundles CUDA support and
# is 800MB-2GB+). The full wheel routinely times out or runs out of memory during a
# Streamlit Community Cloud build.
#
# torch is deliberately left UNPINNED on patch version (only "==2.9.*+cpu") rather
# than an exact version like "2.4.1+cpu": Streamlit Cloud's Python runtime can change
# (e.g. it currently provisions Python 3.13, which has no PyTorch build before 2.9.0),
# and an exact old pin causes "Could not find a version that satisfies the
# requirement" build failures whenever the runtime's Python version moves. If you
# want fully reproducible builds instead, add a runtime.txt / set the Python version
# explicitly in the app's Advanced settings and pin an exact "+cpu" version that
# exists for that Python version.
# =====================================================================================
--extra-index-url https://download.pytorch.org/whl/cpu
